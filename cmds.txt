# randomize pg ids for test

DO $$
DECLARE
    i int;
    pgid text;
BEGIN
    FOR i IN 0..255 LOOP
        pgid := '17.' || to_hex(i);
        EXECUTE format(
            'CREATE TABLE IF NOT EXISTS objects_pg_%s PARTITION OF objects FOR VALUES IN (%L);',
            replace(pgid, '.', '_'),
            pgid
        );
    END LOOP;
END$$;

UPDATE shadow
SET pg_ids = (
  SELECT array_agg('17.' || to_hex(floor(random() * 256)::int))
  FROM generate_series(1, array_length(pg_ids, 1))
);

UPDATE objects
SET pg_id = '17.' || to_hex(floor(random() * 256)::int);
